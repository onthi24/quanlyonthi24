match[2] : ''; } function isYouTubeLink(url) { if (!url) return false; return url.includes('youtube.com') || url.includes('youtu.be'); } function toggleMaximizeVideo() { document.getElementById("main-layout-container").classList.toggle("maximized-mode"); } function updateVideoIframe(videoId, isAutoplay) { const player = document.getElementById('video-player'); const autoplayVal = isAutoplay ? 1 : 0; const targetSrc = `https://www.youtube.com/embed/${videoId}?autoplay=${autoplayVal}&rel=0&showinfo=0&controls=1&playsinline=1&enablejsapi=1`; if (player) player.src = targetSrc; } function initTeachersDataPackage(data) { GROUPED_TEACHERS_DATA = {}; if (data.byTeachers) { GROUPED_TEACHERS_DATA = data.byTeachers; document.getElementById('base-subject-text').innerText = data.subjectTitle || CONFIG.SUBJECT_NAME; } else { const tName = data.sidebarTitle || "Mục lục bài bài giảng (Khoá nền tảng)"; GROUPED_TEACHERS_DATA[tName] = { sidebarTitle: tName, lessons: data.lessons || [] }; document.getElementById('base-subject-text').innerText = data.subjectTitle || CONFIG.SUBJECT_NAME; } renderCustomTeacherLinks(); } // Bổ sung đầy đủ hàm bị thiếu để tránh lỗi hệ thống function renderCustomTeacherLinks() { const container = document.getElementById('custom-teacher-links-list'); if (!container) return; container.innerHTML = ''; Object.keys(GROUPED_TEACHERS_DATA).forEach((tName, index) => { const li = document.createElement('li'); const btn = document.createElement('button'); btn.className = 'course-link-item'; if (index === 0) { btn.className += ' selected-active'; CURRENT_TEACHER_NAME = tName; } btn.innerText = tName; btn.onclick = function() { document.querySelectorAll('.course-link-item').forEach(b => b.classList.remove('selected-active')); btn.classList.add('selected-active'); switchTeacher(tName); }; li.appendChild(btn); container.appendChild(li); }); if (CURRENT_TEACHER_NAME) { switchTeacher(CURRENT_TEACHER_NAME); } } function switchTeacher(tName) { CURRENT_TEACHER_NAME = tName; const teacherData = GROUPED_TEACHERS_DATA[tName]; if (!teacherData) return; document.getElementById('sidebar-title').innerHTML = `${teacherData.sidebarTitle || 'Mục lục'}`; const menuContainer = document.getElementById('menu-container'); if (!menuContainer) return; menuContainer.innerHTML = ''; if (!teacherData.lessons || teacherData.lessons.length === 0) { menuContainer.innerHTML = '
Chưa có bài giảng nào.
'; return; } teacherData.lessons.forEach((lesson, lIdx) => { const rowBlock = document.createElement('div'); rowBlock.className = 'lesson-row-block'; const header = document.createElement('div'); header.className = 'menu-lesson-header'; header.innerText = lesson.lessonTitle; const partsContainer = document.createElement('div'); partsContainer.className = 'lesson-parts-container'; if (lesson.parts && lesson.parts.length > 0) { header.classList.add('has-child'); lesson.parts.forEach((part) => { const item = document.createElement('div'); item.className = 'menu-item'; const nameSpan = document.createElement('span'); nameSpan.className = 'lesson-name'; nameSpan.innerText = part.partTitle; item.appendChild(nameSpan); item.onclick = function() { document.querySelectorAll('.menu-item').forEach(i => i.classList.remove('active')); item.classList.add('active'); document.getElementById('current-lesson-title').innerText = `${lesson.lessonTitle} - ${part.partTitle}`; if (isYouTubeLink(part.linkUrl)) { const yId = extractYouTubeId(part.linkUrl); updateVideoIframe(yId, true); } else if (part.linkUrl) { window.open(part.linkUrl, '_blank'); } }; partsContainer.appendChild(item); }); header.onclick = function() { header.classList.toggle('collapsed'); partsContainer.classList.toggle('hidden'); }; } rowBlock.appendChild(header); rowBlock.appendChild(partsContainer); menuContainer.appendChild(rowBlock); // Tự động kích hoạt bài đầu tiên if (lIdx === 0 && lesson.parts && lesson.parts.length > 0) { const firstItem = partsContainer.querySelector('.menu-item'); if (firstItem) firstItem.click(); } }); } // Tự động chạy khi load trang window.onload = function() { if (typeof STATIC_DATA_STORE !== 'undefined') { initTeachersDataPackage(STATIC_DATA_STORE); } };
